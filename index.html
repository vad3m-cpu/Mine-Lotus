end if; insert into profiles(user_id,name,avatar,pond_bg,lotus_img,pond_public) values(auth.uid(),left(nullif(trim(p_name),''),30),p_avatar,p_bg,p_lotus,coalesce(p_public,true)) on conflict (user_id) do update set name=excluded.name,avatar=excluded.avatar,pond_bg=excluded.pond_bg,lotus_img=excluded.lotus_img,pond_public=excluded.pond_public; end $$; -- เยี่ยมบ่อ: เห็นเฉพาะโน้ตที่เจ้าของเปิดให้เห็น (pub) ไม่ถูกซ่อน (locked) และไม่ใช่โน้ตการนอน create or replace function public.visit_pond(p_code text) returns jsonb language plpgsql security definer set search_path=public as $$ declare p profiles%rowtype; begin if auth.uid() is null then raise exception 'กรุณาเข้าสู่ระบบ'; end if; select * into p from profiles where code=upper(trim(p_code)); if not found or (not p.pond_public and p.user_id<>auth.uid()) then raise exception 'ไม่พบบ่อนี้ หรือเจ้าของปิดไว้'; end if; return jsonb_build_object('code',p.code,'name',coalesce(p.name,'เจ้าของบ่อ'),'avatar',p.avatar,'bg',p.pond_bg,'lotus',p.lotus_img, 'notes',coalesce((select jsonb_agg(jsonb_build_object('id',n.id,'t',n.t,'m',n.m,'x',n.x,'d',n.d,'img',n.img,'song',n.song) order by n.d desc) from (select * from notes where user_id=p.user_id and pub and not locked and t<>'sleep' order by d desc limit 30) n),'[]'::jsonb)); end $$; create or replace function public.random_pond() returns text language sql security definer set search_path=public as $$ select p.code from profiles p where p.pond_public and p.user_id<>auth.uid() and exists(select 1 from notes n where n.user_id=p.user_id and n.pub and not n.locked and n.t<>'sleep') order by random() limit 1 $$; end $$; create or replace function public.catch_lotus() returns table(id bigint,x text,m text,song text,style text,img text,created bigint) language plpgsql security definer set search_path=public as $$ begin return query update floats f set claimed_by=auth.uid() where f.id=(select g.id from floats g where g.to_user is null and g.claimed_by is null and g.from_user<>auth.uid() order by random() limit 1 for update skip locked) returning f.id,f.x,f.m,f.song,f.style,f.img,(extract(epoch from f.created)*1000)::bigint;
